===== Lua Implementation (Barkskin.lua) =====
<code lua>
local RPD  = require "scripts/lib/commonClasses"
local buff = require "scripts/lib/buff"

return buff.init{
    desc  = function ()
        return {
            icon          = 24, -- BuffIndicator.BARKSKIN
            name          = "BarkskinBuff_Name",
            info          = "BarkskinBuff_Info",
        }
    end,

    act = function(self, buff)
        local target = buff.target
        if target:isAlive() then
            buff:spend(1) -- Actor.TICK
            local lvl = buff:level() - 1
            buff:level(lvl)
            if lvl <= 0 then
                buff:detach()
            end
        else
            buff:detach()
        end
    end,

    drBonus = function(self, buff)
        return buff:level()
    end
}
</code>

===== Mechanics =====
  * **Effect**: Increases DR (Damage Reduction) by amount equal to current buff level
  * **Duration**: Buff level decreases by 1 each turn (Actor.TICK); expires when level reaches 0
  * **Initial Level**: Set by the source applying the buff (see Sources below)
  * **DR Bonus**: Equal to current buff level (via `drBonus()` method)
  * **Icon**: BuffIndicator.BARKSKIN (icon ID 24)
